   208→function MarqueeRow({
   209→  items,
   210→  animation,
   211→  colorOffset,
   212→}: {
   213→  items: MarqueeItem[]
   214→  animation: string
   215→  colorOffset: number
   216→}) {
   217→  return (
   218→    <div className="group overflow-hidden [mask-image:linear-gradient(to_right,transparent,black_12%,black_88%,transparent)]">
   219→      <div
   220→        className={`flex w-max gap-3 pr-3 will-change-transform group-hover:[animation-play-state:paused] motion-reduce:animate-none ${animation}`}
   221→      >
   222→        {[0, 1].map((copy) =>
   223→          items.map((item, i) => {
   224→            const color = PILL_COLORS[(i + colorOffset) % PILL_COLORS.length]
   225→            const Icon = item.icon
   226→            return (
   227→              <span
   228→                key={`${copy}-${item.label}`}
   229→                aria-hidden={copy === 1}
   230→                className={`flex shrink-0 items-center gap-2 rounded-full px-4 py-2.5 text-[13px] font-semibold text-[#6B5544] ${color.bg} ${color.shadow}`}
   231→              >
   232→                <Icon className={`size-4 ${color.icon}`} />
   233→                {item.label}
   234→              </span>
   235→            )
   236→          }),
   237→        )}
   238→      </div>
   239→    </div>
   240→  )
   241→}
