   116→async function getTarget(): Promise<DbTarget> {
   117→  const turso = tursoClient();
   118→  if (turso) {
   119→    const fresh = Date.now() - tursoProbeAt > 600_000;
   120→    if (tursoMigrated && !fresh) {
   121→      return { client: turso, source: "turso" };
   122→    }
   123→    /* v3.02 · PROBE — SELECT 1 murah; gagal berarti kredensial/DB bermasalah.
   124→       Alasan disimpan agar GET/POST bisa melaporkan APA yang salah
   125→       (401 = token, 404 = DB terhapus, jaringan, dst.) alih-alih 502 bisu. */
   126→    try {
   127→      await turso.execute("SELECT 1 AS ok");
   128→    } catch (error) {
   129→      tursoDownReason = errText(error);
   130→      console.error("[/api/rutina] Turso tak terjangkau — fallback lokal:", tursoDownReason);
   131→      return localTarget(tursoDownReason);
   132→    }
   133→    tursoDownReason = null;
   134→    tursoProbeAt = Date.now();
   135→    /* v2.35 · pastikan kolom Atomic Habits hadir di database produksi.
   136→       v2.58 · tabel Bugar juga dibuat idempoten di Turso (CREATE IF NOT
   137→       EXISTS aman diputar ulang tiap boot proses). */
   138→    if (!tursoMigrated) {
   139→      for (const sql of DDL) {
   140→        try { await turso.execute(sql); } catch (error) {
   141→          /* v3.04.1 · "sudah ada" wajar diabaikan; kegagalan lain (izin,
   142→             read-only) kini TERCATAT — dulu ditelan dan baru meledak sebagai
   143→             502 membingungkan di query pertama. */
   144→          const m = errText(error);
   145→          if (!/already exists|duplicate column/i.test(m)) console.error("[/api/rutina] DDL Turso:", m);
   146→        }
   147→      }
   148→      await migrateColumns(turso);
   149→      tursoMigrated = true;
   150→    }
   151→    return { client: turso, source: "turso" };
   152→  }
   153→  return localTarget(null);
   154→}

   156→/* v3.02 · SQLite lokal — dua anak tangga: direktori kerja (server biasa,
   157→   data abadi) lalu /tmp (lambda Vercel read-only, data per-instance).
   158→   `volatile` memberi tahu klien bahwa penyimpanan ini bisa menguap saat
   159→   instance dingin — klien lalu TIDAK membersihkan jurnalnya sehingga
   160→   seluruh riwayat tetap terdorong ulang ke server berikutnya. */
   161→const localFailed = new Set<string>(); /* v3.04.1 · path yang gagal SIAP (db korup / disk penuh) — dulu kegagalan skema hanya terdeteksi async sehingga loop tak pernah mencoba anak tangga /tmp */
   162→async function localTarget(reason: string | null): Promise<DbTarget> {
   163→  if (!cachedLocal) {
   164→    const dirs: [string, boolean][] = [
   165→      [LOCAL_DB, false],
   166→      [TMP_DB, true],
   167→    ];
   168→    let lastError: unknown = null;
   169→    for (const [path, volatile] of dirs) {
   170→      if (localFailed.has(path)) continue;
   171→      try {
   172→        mkdirSync(dirname(path), { recursive: true });
   173→        const client = createClient({ url: `file:${path}` });
   174→        const ready = ensureLocalSchema(client).catch((error) => {
   175→          console.error("[/api/rutina] gagal menyiapkan database lokal:", error);
   176→          localFailed.add(path);
   177→          cachedLocal = null;
   178→          throw error;
   179→        });
   180→        cachedLocal = { client, ready, volatile };
   181→        break;
   182→      } catch (error) {
   183→        lastError = error; /* read-only/EROFS → coba anak tangga berikutnya */
   184→        localFailed.add(path);
   185→        cachedLocal = null;
   186→      }
   187→    }
   188→    if (!cachedLocal) {
   189→      throw lastError ?? new Error("Tak ada penyimpanan lokal yang tersedia.");
   190→    }
   191→  }
   192→  await cachedLocal.ready; /* DDL & seed selesai SEBELUM klien dipakai — tanpa ini query pertama berlomba dengan pembuatan tabel */
   193→  return {
   194→    client: cachedLocal.client,
   195→    source: "local",
   196→    volatile: cachedLocal.volatile,
   197→    reason: reason ?? undefined,
   198→  };
   199→}

   262→async function ensureLocalSchema(client: Client): Promise<void> {
   263→  /* v2.60 · hardening SQLite pratinjau — audit "koneksi ke database aman
   264→     dan sehat ya?" menemukan file db memakai journal_mode=delete bawaan:
   265→     satu penulis mengunci seluruh file → pembacaan GET saat sinkronisasi
   266→     POST berjalan bisa kena SQLITE_BUSY. Dua pragma permanen memperbaikinya
   267→     tanpa mengubah semantik transaksi:
   268→     · WAL — pembaca & penulis hidup bersamaan; file -wal/-shm akan muncul
   269→       di samping db/rutina-preview.db (normal, bukan kebocoran).
   270→     · busy_timeout 5 dtk — penulis kedua menunggu antrean, bukan gagal.
   271→     klien lokal adalah singleton per proses, jadi pragma sekali di boot
   272→     berlaku untuk seluruh umur server; idempoten & aman diulang. */
   273→  try {
   274→    await client.execute("PRAGMA journal_mode=WAL");
   275→    await client.execute("PRAGMA busy_timeout=5000");
   276→    /* v2.64 · rampingkan WAL peninggalan saat boot — audit menemukan file
   277→       -wal membengkak 25× ukuran db (4 MB vs 160 KB) karena tiap POST
   278→       menulis ulang dua tabel penuh dalam ~34 autocommit terpisah; kini
   279→       tulisan dibatch (lihat applyOverlay) + checkpoint di boot & pasca-POST. */
   280→    try { await client.execute("PRAGMA wal_checkpoint(TRUNCATE)"); } catch { /* pembaca aktif — nanti saja */ }
   281→  } catch { /* modul read-only / fs aneh → lanjut dengan perilaku lama */ }
   282→  for (const sql of DDL) await client.execute(sql);
   283→  await migrateColumns(client);
   284→  const res = await client.execute(`SELECT COUNT(*) AS n FROM "Habit"`);
   285→  const n = Number((res.rows[0] as { n?: unknown } | undefined)?.n ?? 0);
   286→  if (n === 0) await seedFromSnapshot(client);
   287→}

   292→let snapshotCache: Record<string, unknown> | null | undefined;
   293→function loadSnapshot(): Record<string, unknown> | null {
   294→  if (snapshotCache !== undefined) return snapshotCache;
   295→  snapshotCache = null;
   296→  try {
   297→    if (!existsSync(HTML_SNAPSHOT)) return snapshotCache;
   298→    const html = readFileSync(HTML_SNAPSHOT, "utf8");
   299→    const m = html.match(/<script[^>]*id="dbSnapshot"[^>]*>([\s\S]*?)<\/script>/);
   300→    if (!m) return snapshotCache;
   301→    const parsed = JSON.parse(m[1]) as { ok?: boolean; data?: Record<string, unknown> };
   302→    if (parsed?.ok && parsed?.data) snapshotCache = parsed.data;
   303→  } catch {
   304→    snapshotCache = null;
   305→  }
   306→  return snapshotCache;
   307→}

   432→async function columnsOf(client: Client, table: string): Promise<Set<string>> {
   433→  const hit = colCache.get(table);
   434→  if (hit) return hit;
   435→  let cols: Set<string>;
   436→  try {
   437→    const res = await client.execute(`PRAGMA table_info("${table}")`);
   438→    cols = new Set(res.rows.map((r) => str((r as { name?: unknown }).name)));
   439→    if (cols.size === 0) cols = new Set(FALLBACK_COLS[table] ?? []);
   440→  } catch {
   441→    cols = new Set(FALLBACK_COLS[table] ?? []);
   442→  }
   443→  colCache.set(table, cols);
   444→  return cols;
   445→}

   447→/* INSERT yang hanya memakai kolom yang benar-benar ada di tabel target —
   448→   mencocokkan diri dengan skema asli apa pun bentuknya (mis. kolom waktu
   449→   opsional createdAt/updatedAt di Turso legacy ikut terisi bila tersedia).
   450→   v2.64 · buildInsert murni (sinkron) — dipakai merangkai pernyataan utk
   451→   client.batch di applyOverlay; jalur penyemaian memakai SQL eksplisit. */
   452→function buildInsert(cols: Set<string>, table: string, row: Record<string, unknown>): { sql: string; args: InValue[] } | null {
   453→  const names = Object.keys(row).filter((c) => cols.has(c));
   454→  if (cols.has("createdAt") && !names.includes("createdAt")) names.push("createdAt");
   455→  if (cols.has("updatedAt") && !names.includes("updatedAt")) names.push("updatedAt");
   456→  if (!names.length) return null;
   457→  const now = new Date().toISOString();
   458→  const args: InValue[] = names.map((n) => inVal(n in row ? (row[n] ?? null) : now));
   459→  const sql = `INSERT INTO "${table}" (${names.map((n) => `"${n}"`).join(", ")}) VALUES (${names.map(() => "?").join(", ")})`;
   460→  return { sql, args };
   461→}

   754→const MEALS = ["pagi", "siang", "malam", "camilan"] as const;
   755→function sanitizeSet(v: unknown): SetRow {
   756→  const o = asRecord(v);
   757→  return {
   758→    id: str(o.id),
   759→    exercise: str(o.exercise).slice(0, 60),
   760→    reps: Math.round(clamp(num(o.reps), 0, 500)),
   761→    weight: clamp(num(o.weight), 0, 1000),
   762→  };
   763→}
   764→function sanitizeWorkout(v: unknown): WorkoutRow {
   765→  const o = asRecord(v);
   766→  const kindRaw = str(o.kind);
   767→  const kind = (KINDS as readonly string[]).includes(kindRaw) ? kindRaw : "beban";
   768→  const hasDist = kind !== "beban" && o.distance != null && num(o.distance) > 0;
   769→  return {
   770→    id: str(o.id),
   771→    date: str(o.date).slice(0, 10),
   772→    kind,
   773→    duration: Math.round(clamp(num(o.duration), 0, 1440)),
   774→    distance: hasDist ? clamp(num(o.distance), 0.1, 500) : null,
   775→    note: str(o.note).slice(0, 200),
   776→    sets: Array.isArray(o.sets)
   777→      ? (o.sets as unknown[]).map(sanitizeSet).filter((s) => s.id && s.exercise && s.reps > 0 && s.weight > 0)
   778→      : [],
   779→  };
   780→}
   781→function sanitizeFood(v: unknown): FoodRow {
   782→  const o = asRecord(v);
   783→  const mealRaw = str(o.meal);
   784→  const meal = (MEALS as readonly string[]).includes(mealRaw) ? mealRaw : "camilan";
   785→  return {
   786→    id: str(o.id),
   787→    date: str(o.date).slice(0, 10),
   788→    meal,
   789→    name: str(o.name).slice(0, 80),
   790→    kcal: Math.round(clamp(num(o.kcal), 0, 20000)),
   791→    protein: clamp(num(o.protein), 0, 2000),
   792→    carb: clamp(num(o.carb), 0, 2000),
   793→    fat: clamp(num(o.fat), 0, 2000),
   794→  };
   795→}
   796→function sanitizeBody(v: unknown): BodyRow {
   797→  const o = asRecord(v);
   798→  return {
   799→    date: str(o.date).slice(0, 10),
   800→    water: Math.round(clamp(num(o.water), 0, 40)),
   801→    weight: o.weight != null && num(o.weight) > 0 ? clamp(num(o.weight), 20, 400) : null,
   802→    recover: Math.round(clamp(num(o.recover), 0, 5)),
   803→  };
   804→}
   805→function strList(v: unknown): string[] {
   806→  return Array.isArray(v) ? v.map((x) => str(x)).filter(Boolean) : [];
   807→}
   808→
   809→function sanitizeOverlay(input: unknown): Overlay | null {
   810→  const o = asRecord(input);
   811→  if (typeof o.v !== "number") return null;
   812→  const txList = (v: unknown): TxRow[] => (Array.isArray(v) ? v.map(sanitizeTx).filter((t) => t.id && t.date) : []);
   813→  const budList = (v: unknown): BudRow[] => (Array.isArray(v) ? v.map(sanitizeBud).filter((b) => b.category && b.month) : []);
   814→  const habitList = (v: unknown): HabitRow[] => (Array.isArray(v) ? v.map(sanitizeHabit).filter((h) => h.id && h.name) : []);
   815→  const srcList = (v: unknown): SrcRow[] => (Array.isArray(v) ? v.map(sanitizeSrc).filter((s) => s.id && s.name) : []);
   816→  const txMap: Record<string, TxRow> = {};
   817→  for (const [k, v] of Object.entries(asRecord(o.txEdit))) {
   818→    const t = sanitizeTx(v);
   819→    if (k && t.date) txMap[k] = t;
   820→  }
   821→  const budMap: Record<string, BudRow> = {};
   822→  for (const [k, v] of Object.entries(asRecord(o.budEdit))) {
   823→    const b = sanitizeBud(v);
   824→    if (k && b.category && b.month) budMap[k] = b;
   825→  }
   826→  const habitMap = asRecord(o.logAdd);
   827→  const logAdd: Record<string, number | true> = {};
   828→  for (const k of Object.keys(habitMap)) {
   829→    const v = habitMap[k];
   830→    logAdd[k] = typeof v === "number" ? num(v) : true;
   831→  }
   832→  const logDel: Record<string, true> = {};
   833→  for (const k of Object.keys(asRecord(o.logDel))) logDel[k] = true;
   834→  const catRename: Record<string, string> = {};
   835→  for (const [k, v] of Object.entries(asRecord(o.catRename))) {
   836→    if (k && str(v) && str(v) !== k) catRename[k] = str(v).slice(0, 40);
   837→  }
   838→  /* v2.35 · meta kebiasaan + identitas pengguna; v2.37 · meta diperluas
   839→     (nama/emoji/kategori/jam/grup) — hanya kunci yang HADIR yang disimpan,
   840→     null berarti "kosongkan" (lepas grup / hapus mini/jangkar). Kolom
   841→     nullable diuji dengan "in" (bukan typeof) supaya null eksplisit dari
   842→     klien tetap sampai sebagai perintah kosongkan. */
   843→  const habitMeta: Record<string, HabitMetaRow> = {};
   844→  for (const [k, v] of Object.entries(asRecord(o.habitMeta))) {
   845→    if (!k) continue;
   846→    const m = asRecord(v);
   847→    const e: HabitMetaRow = {};
   848→    if (typeof m.name === "string" && str(m.name)) e.name = str(m.name).slice(0, 80);
   849→    if ("emoji" in m) e.emoji = m.emoji ? str(m.emoji).slice(0, 40) : null;
   850→    if (typeof m.category === "string" && str(m.category)) e.category = str(m.category).slice(0, 40);
   851→    if ("reminder" in m) e.reminder = m.reminder ? str(m.reminder).slice(0, 16) : null;
   852→    if ("groupId" in m) e.groupId = m.groupId ? str(m.groupId).slice(0, 64) : null;
   853→    if ("mini" in m) e.mini = m.mini ? str(m.mini).slice(0, 80) : null;
   854→    if ("afterId" in m) e.afterId = m.afterId ? str(m.afterId).slice(0, 64) : null;
   855→    /* v2.54 · jeda runtun aman — CSV tanggal YYYY-MM-DD; token tak valid
   856→       dibuang, duplikat dirapatkan, urut, dibatasi 260 entri (5 tahun
   857→       jeda mingguan). "" / null = kosongkan kolom. */
   858→    if ("restDays" in m) {
   859→      const rawRest = m.restDays;
   860→      if (rawRest === null || rawRest === undefined || rawRest === "") e.restDays = null;
   861→      else {
   862→        const tokens = (Array.isArray(rawRest) ? rawRest.map((x) => str(x)) : String(rawRest).split(","))
   863→          .map((t) => t.trim()).filter((t) => /^\d{4}-\d{2}-\d{2}$/.test(t));
   864→        const uniq = [...new Set(tokens)].sort().slice(0, 260);
   865→        e.restDays = uniq.length ? uniq.join(",") : null;
   866→      }
   867→    }
   868→    if (Object.keys(e).length) habitMeta[k] = e;
   869→  }
   870→  const identityText = typeof o.identityText === "string" ? o.identityText.slice(0, 120) : null;
   871→  /* v2.39 · check-in harian: peta tanggal → {mood,energy,sleep}; nilai
   872→     diklem ke rentang wajar (mood/energi 1–5, tidur 0–16 jam), tanggal
   873→     dipotong ke bentuk YYYY-MM-DD; tanggal tak valid dibuang. */
   874→  const dailyLog: Record<string, DailyLogRow> = {};
   875→  for (const [k, v] of Object.entries(asRecord(o.dailyLog))) {
   876→    const date = k.slice(0, 10);
   877→    const m = asRecord(v);
   878→    if (!/^\d{4}-\d{2}-\d{2}$/.test(date) || !m) continue;
   879→    dailyLog[date] = {
   880→      date,
   881→      mood: Math.min(5, Math.max(0, Math.round(num(m.mood)))),
   882→      energy: Math.min(5, Math.max(0, Math.round(num(m.energy)))),
   883→      sleep: Math.min(16, Math.max(0, num(m.sleep))),
   884→    };
   885→  }
   886→  return {
   887→    txAdd: txList(o.txAdd), txEdit: txMap, txDel: strList(o.txDel),
   888→    budAdd: budList(o.budAdd), budEdit: budMap, budDel: strList(o.budDel),
   889→    habitAdd: habitList(o.habitAdd), habitDel: strList(o.habitDel),
   890→    habitMeta, identityText,
   891→    logAdd, logDel,
   892→    srcAdd: srcList(o.srcAdd), srcEdit: (() => {
   893→      const m: Record<string, SrcRow> = {};
   894→      for (const [k, v] of Object.entries(asRecord(o.srcEdit))) {
   895→        const s = sanitizeSrc(v);
   896→        if (k && s.name) m[k] = s;
   897→      }
   898→      return m;
   899→    })(), srcDel: strList(o.srcDel),
   900→    catRename, catDel: strList(o.catDel),
   901→    dailyLog,
   902→    /* v2.58 · tab Bugar — jurnal latihan (set tertanam), asupan, tubuh,
   903→       dan sasaran (JSON teks dibatasi 2000 karakter). */
   904→    fitWAdd: (Array.isArray(o.fitWAdd) ? (o.fitWAdd as unknown[]).map(sanitizeWorkout).filter((w) => w.id && w.date) : []),
   905→    fitWDel: strList(o.fitWDel),
   906→    fitFAdd: (Array.isArray(o.fitFAdd) ? (o.fitFAdd as unknown[]).map(sanitizeFood).filter((f) => f.id && f.date) : []),
   907→    fitFDel: strList(o.fitFDel),
   908→    bodyLog: (() => {
   909→      const m: Record<string, BodyRow> = {};
   910→      for (const [k, v] of Object.entries(asRecord(o.bodyLog))) {
   911→        const date = k.slice(0, 10);
   912→        if (!/^\d{4}-\d{2}-\d{2}$/.test(date)) continue;
   913→        const b = sanitizeBody({ ...(asRecord(v) as Record<string, unknown>), date });
   914→        m[date] = b;
   915→      }
   916→      return m;
   917→    })(),
   918→    fitness: typeof o.fitness === "string" && o.fitness.trim().startsWith("{") ? o.fitness.slice(0, 2000) : null,
   919→    /* v3.02 · coach Bugar — kesiapan (tanggal → nilai diklem) & target
   920→       gerakan (nama → beban/reps); keduanya upsert idempoten by key. */
   921→    readiness: (() => {
   922→      const m: Record<string, ReadinessRow> = {};
   923→      for (const [k, v] of Object.entries(asRecord(o.readiness))) {
   924→        const date = k.slice(0, 10);
   925→        if (!/^\d{4}-\d{2}-\d{2}$/.test(date)) continue;
   926→        const r = asRecord(v);
   927→        m[date] = {
   928→          date,
   929→          sleep: clamp(num(r.sleep), 0, 24), /* v3.04.1 · disamakan dgn klien (input 0–24 jam) — dulu clamp 16 memotong nilai diam-diam pasca-reload */
   930→          doms: Math.round(clamp(num(r.doms), 0, 5)),
   931→          stress: Math.round(clamp(num(r.stress), 0, 5)),
   932→          energy: Math.round(clamp(num(r.energy), 0, 5)),
   933→        };
   934→      }
   935→      return m;
   936→    })(),
   937→    fitTgt: (() => {
   938→      const m: Record<string, ExTargetRow> = {};
   939→      for (const [k, v] of Object.entries(asRecord(o.fitTgt))) {
   940→        const exercise = str(k).slice(0, 60);
   941→        if (!exercise) continue;
   942→        const r = asRecord(v);
   943→        /* v3.04.1 · weight/reps 0 dilewati — dulu weight di-clamp jadi 1 kg
   944→           sehingga lahir baris target sampah {1 kg × 1 rep} */
   945→        if (num(r.weight) <= 0 || num(r.reps) <= 0) continue;
   946→        m[exercise] = { exercise, weight: clamp(num(r.weight), 1, 1000), reps: Math.round(clamp(num(r.reps), 1, 500)) };
   947→      }
   948→      return m;
   949→    })(),
   950→  };
   951→}

   953→/* kunci jurnal "habitId|date" dan "kategori|bulan" dipecah pada '|' pertama */
   954→function splitKey(key: string): [string, string] {
   955→  const i = key.indexOf("|");
   956→  return i < 0 ? [key, ""] : [key.slice(0, i), key.slice(i + 1)];
   957→}

   959→export async function POST(request: Request) {
   960→  /* v3.04.1 · batas ukuran ditegakkan pada ISI body — content-length bisa
   961→     absen (chunked) atau non-numerik sehingga cek header sendiri tembus. */
   962→  const hdrSize = Number(request.headers.get("content-length") ?? NaN);
   963→  if (hdrSize > MAX_BODY_BYTES) {
   964→    return NextResponse.json({ ok: false, error: "Lapisan catatan terlalu besar." }, { status: 413, headers: NO_STORE });
   965→  }
   966→  let raw = "";
   967→  try { raw = await request.text(); } catch {
   968→    return NextResponse.json({ ok: false, error: "Body tidak terbaca." }, { status: 400, headers: NO_STORE });
   969→  }
   970→  if (raw.length > MAX_BODY_BYTES) {
   971→    return NextResponse.json({ ok: false, error: "Lapisan catatan terlalu besar." }, { status: 413, headers: NO_STORE });
   972→  }
   973→  let body: unknown;
   974→  try {
   975→    body = JSON.parse(raw);
   976→  } catch {
   977→    return NextResponse.json({ ok: false, error: "Body bukan JSON yang valid." }, { status: 400, headers: NO_STORE });
   978→  }
   979→  const b = asRecord(body);
   980→  const overlay = sanitizeOverlay(b.overlay);
   981→  if (!overlay) {
   982→    return NextResponse.json({ ok: false, error: "Struktur lapisan catatan tidak dikenal." }, { status: 400, headers: NO_STORE });
   983→  }
   984→
   985→  let target: DbTarget;
   986→  try {
   987→    target = await getTarget();
   988→  } catch {
   989→    return NextResponse.json({ ok: false, error: "Sumber data belum siap." }, { status: 503, headers: NO_STORE });
   990→  }
   991→
   992→  try {
   993→    const result = await applyOverlay(
   994→      target.client,
   995→      overlay,
   996→      Array.isArray(b.budgets) ? (b.budgets as unknown[]) : null,
   997→      Array.isArray(b.finCats) ? (b.finCats as unknown[]) : null,
   998→    );
   999→    /* v3.04.1 · data berubah → revisi naik DI DATABASE: ETag baru, GET
  1000→       berikutnya (instance mana pun) mengunduh isinya. */
  1001→    if (Object.values(result.applied).reduce((a, b) => a + b, 0) > 0) {
  1002→      await bumpRev(target.client);
  1003→    }
  1004→    /* v2.64 · WAL lokal dirampingkan pasca-POST — tulisan terbatch kini
  1005→       kecil, checkpoint TRUNCATE menjaga -wal tak menumpuk antar sesi. */
  1006→    if (target.source === "local") {
  1007→      try { await target.client.execute("PRAGMA wal_checkpoint(TRUNCATE)"); } catch { /* pembaca aktif — nanti saja */ }
  1008→    }
  1009→    return NextResponse.json({ ok: true, source: target.source, volatile: target.volatile === true, reason: target.reason ?? null, ...result }, { headers: NO_STORE });
  1010→  } catch (error) {
  1011→    console.error("[/api/rutina] gagal menerapkan catatan:", error);
  1012→    return NextResponse.json({ ok: false, error: "Gagal menyimpan ke database." }, { status: 502, headers: NO_STORE });
  1013→  }
  1014→}

  1016→type Stmt = { sql: string; args: InValue[] };
  1017→async function applyOverlay(
  1018→  client: Client,
  1019→  ov: Overlay,
  1020→  budgets: unknown[] | null,
  1021→  finCats: unknown[] | null,
  1022→): Promise<{ applied: Record<string, number>; failed: { op: string; error: string }[] }> {
  1023→  const applied: Record<string, number> = {};
  1024→  const failed: { op: string; error: string }[] = [];
  1025→  /* v2.64 · SEKSI-BATCH — dulu setiap pernyataan dieksekusi await-sendiri
  1026→     (autocommit terpisah): satu POST memutar ~34 transaksi implicit, diukur
  1027→     6× lebih lambat dari satu BEGIN/COMMIT di disk lokal — dan di Turso
  1028→     tiap pernyataan = satu HTTP round-trip beruntun. Kini tiap seksi
  1029→     dirangkai penuh lalu dikirim SEKALI lewat client.batch(..., "write")
  1030→     = satu transaksi tulis atomik (gagal di tengah → seluruh seksi
  1031→     tergulung, jurnal tetap di klien & diputar ulang idempoten — lebih
  1032→     aman dari laku lama yang bisa setengah-terpasang). */
  1033→  const [txCols, budCols, habitCols, srcCols, logCols, workoutCols, wsetCols, foodCols] =
  1034→    await Promise.all([
  1035→      columnsOf(client, "Transaction"), columnsOf(client, "WeeklyBudget"),
  1036→      columnsOf(client, "Habit"), columnsOf(client, "FundSource"),
  1037→      columnsOf(client, "HabitLog"), columnsOf(client, "Workout"),
  1038→      columnsOf(client, "WorkoutSet"), columnsOf(client, "FoodLog"),
  1039→    ]);
  1040→  const push = (S: Stmt[], cols: Set<string>, table: string, row: Record<string, unknown>) => {
  1041→    const st = buildInsert(cols, table, row);
  1042→    if (st) S.push(st);
  1043→  };
  1044→  const batchRun = async (op: string, build: (S: Stmt[]) => void) => {
  1045→    const S: Stmt[] = [];
  1046→    try {
  1047→      build(S);
  1048→      if (S.length) await client.batch(S, "write");
  1049→      if (S.length) applied[op] = S.length;
  1050→    } catch (error) {
  1051→      failed.push({ op, error: String((error as Error)?.message ?? error).slice(0, 160) });
  1052→    }
  1053→  };
  1054→  const run = async (op: string, fn: () => Promise<unknown>) => {
  1055→    try {
  1056→      await fn();
  1057→      applied[op] = (applied[op] ?? 0) + 1;
  1058→    } catch (error) {
  1059→      failed.push({ op, error: String((error as Error)?.message ?? error).slice(0, 160) });
  1060→    }
  1061→  };
  1062→  const mapCat = (c: string): string => ov.catRename[c] || c;
  1063→
  1064→  /* 1 · ganti nama kategori pada transaksi. (Daftar kategori & anggaran
  1065→        diganti penuh dari daftar final klien di bawah — tak perlu di-update.) */
  1066→  await batchRun("catRename", (S) => {
  1067→    for (const [oldName, newName] of Object.entries(ov.catRename)) {
  1068→      S.push({ sql: `UPDATE "Transaction" SET "category"=? WHERE "category"=?`, args: [newName, oldName] });
  1069→    }
  1070→  });
  1071→
  1072→  /* 2 · transaksi: hapus → tambal → tambah (tiap jalur satu batch) */
  1073→  await batchRun("txDel", (S) => {
  1074→    for (const id of ov.txDel) S.push({ sql: `DELETE FROM "Transaction" WHERE "id"=?`, args: [id] });
  1075→  });
  1076→  await batchRun("txEdit", (S) => {
  1077→    for (const [id, t] of Object.entries(ov.txEdit)) {
  1078→      if (!id || !t.date) continue;
  1079→      /* v2.37 · recurrence hanya ikut bila kuncinya hadir (jurnal lama tanpa
  1080→         kolom ini tak akan menimpa nilai berulang yang tersimpan) */
  1081→      const hasRec = t.recurrence !== undefined;
  1082→      S.push({
  1083→        sql: hasRec
  1084→          ? `UPDATE "Transaction" SET "type"=?,"amount"=?,"category"=?,"sourceId"=?,"description"=?,"date"=?,"recurrence"=? WHERE "id"=?`
  1085→          : `UPDATE "Transaction" SET "type"=?,"amount"=?,"category"=?,"sourceId"=?,"description"=?,"date"=? WHERE "id"=?`,
  1086→        args: hasRec
  1087→          ? [t.type, t.amount, mapCat(t.category), t.sourceId, t.description, t.date, t.recurrence ?? null, id]
  1088→          : [t.type, t.amount, mapCat(t.category), t.sourceId, t.description, t.date, id],
  1089→      });
  1090→    }
  1091→  });
  1092→  await batchRun("txAdd", (S) => {
  1093→    for (const t of ov.txAdd) {
  1094→      S.push({ sql: `DELETE FROM "Transaction" WHERE "id"=?`, args: [t.id] });
  1095→      push(S, txCols, "Transaction", {
  1096→        id: t.id, type: t.type, amount: t.amount, category: mapCat(t.category),
  1097→        sourceId: t.sourceId, description: t.description, date: t.date,
  1098→        recurrence: t.recurrence ?? null, /* v2.37 */
  1099→      });
  1100→    }
  1101→  });
  1102→
  1103→  /* 2b · v3.04.1 · ANGGARAN & KATEGORI dari JURNAL — jalur pemulihan:
  1104→        saat server volatile klien MEMPERTAHANKAN jurnal namun flag daftar
  1105→        final (budgets/finCats) bisa sudah bersih; tanpa seksi ini edit
  1106→        anggaran & penghapusan kategori selama outage hilang permanen begitu
  1107→        Turso pulih (jurnal direplay TANPA daftar final). Idempoten: budEdit
  1108→        menghapus kunci lama lalu upsert baris baru (rename), budAdd upsert
  1109→        by kategori|bulan, budDel/catDel = DELETE. Ditempatkan SEBELUM
  1110→        daftar final supaya daftar final tetap menang saat keduanya hadir. */
  1111→  await batchRun("budJournal", (S) => {
  1112→    const seen = new Set<string>();
  1113→    const put = (b: BudRow) => {
  1114→      const key = `${b.category}|${b.month}`;
  1115→      if (seen.has(key)) return;
  1116→      seen.add(key);
  1117→      S.push({ sql: `INSERT OR REPLACE INTO "WeeklyBudget"("category","month","amount") VALUES (?,?,?)`, args: [mapCat(b.category), b.month, b.amount] });
  1118→    };
  1119→    for (const [oldKey, b] of Object.entries(ov.budEdit)) {
  1120→      const [oc, om] = splitKey(oldKey);
  1121→      if (!oc || !om || !b.category || !b.month) continue;
  1122→      S.push({ sql: `DELETE FROM "WeeklyBudget" WHERE "category"=? AND "month"=?`, args: [oc, om] });
  1123→      put(b);
  1124→    }
  1125→    for (const b of ov.budAdd) if (b.category && b.month) put(b);
  1126→  });
